// Archival only. Insert inside the existing DEBUG-only DeveloperSettingsSection.swift.
// This is the helper actually used by the local, uncommitted hooks.
nonisolated enum SuspectProbe {
    static func seconds(_ flag: String) -> Double {
        let args = ProcessInfo.processInfo.arguments
        guard let i = args.firstIndex(of: "--suspect-" + flag), args.indices.contains(i + 1) else {
            let path = FileManager.default.temporaryDirectory.appendingPathComponent("suspect-" + flag)
            return (try? String(contentsOf: path, encoding: .utf8)).flatMap { Double($0.trimmingCharacters(in: .whitespacesAndNewlines)) } ?? 0
        }
        return Double(args[i + 1]) ?? 0
    }
    static func trace(_ message: String) {
        let url = FileManager.default.temporaryDirectory.appendingPathComponent("suspects-trace.log")
        let data = Data("\(Date().timeIntervalSince1970) \(message)\n".utf8)
        if !FileManager.default.fileExists(atPath: url.path) { FileManager.default.createFile(atPath: url.path, contents: nil) }
        if let handle = try? FileHandle(forWritingTo: url) {
            defer { try? handle.close() }
            try? handle.seekToEnd()
            try? handle.write(contentsOf: data)
        }
    }
    static func pause(_ flag: String) async {
        let delay = seconds(flag)
        guard delay > 0 else { return }
        trace("pause \(flag) begin")
        try? await Task.sleep(for: .seconds(delay))
        trace("pause \(flag) end cancelled=\(Task.isCancelled)")
    }
}
